def df_to_list(df):
    headers = list(df.columns.values)
    list_out = []
    for header in headers:
        df[header].str.strip()
        col = df[header].tolist()
        for each in col:
            if str(each).isalpha() and len(str(each)) > 2:
                list_out.append(each)
            else:
                pass
    return(list_out)
